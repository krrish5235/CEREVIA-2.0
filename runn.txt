# ╔═══════════════════════════════════════════════════════════════╗
# ║              CEREVIA — How to Run the Project               ║
# ╠═══════════════════════════════════════════════════════════════╣
# ║  3 Microservices need to run simultaneously:                ║
# ║    1. C++ Core Backend       (Port 5000)                    ║
# ║    2. Python AI Chatbot      (Port 5001)                    ║
# ║    3. Java Analytics Service (Port 8080)                    ║
# ╚═══════════════════════════════════════════════════════════════╝

# ─── Step 1: Build & Run C++ Backend (Port 5000) ────────────────
cmake --build backend/build --config Debug
.\backend\build\Debug\backend.exe

# ─── Step 2: Run Python AI Chatbot (Port 5001) ──────────────────
# First time only: install dependencies
pip install -r ai_chatbot/requirements.txt

# Set Gemini API key (get free key from https://aistudio.google.com/apikey)
set GEMINI_API_KEY=your-api-key-here

# Run chatbot
python ai_chatbot/app.py

# ─── Step 3: Run Java Analytics Service (Port 8080) ─────────────
# First time only: build with Maven
cd analytics
mvn clean package -DskipTests
cd ..

# Run analytics
java -jar analytics/target/analytics-1.0.0.jar

# ─── Step 4: Open Frontend ──────────────────────────────────────
# Open frontend/index.html in browser (or use VS Code Live Server)
